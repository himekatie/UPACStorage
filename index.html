Connect to Google Sheets

Shares live data across all users. Do this once.

1. Create a Google Sheet with columns: Name, Location, Category, Quantity, Notes
2. Go to Extensions → Apps Script, paste the script below
3. Click Deploy → New deployment → Web app → Execute as: Me → Who has access: Anyone → Deploy
4. Copy the /exec URL and paste it here.

Show Apps Script code to paste → 
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1).map((r, i) => ({
    id: i + 2,
    ...Object.fromEntries(headers.map((h, j) => [h.toLowerCase(), r[j]]))
  }));
  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const p = JSON.parse(e.postData.contents);
  if (p.action === 'add')
    sheet.appendRow([p.name, p.location, p.category, p.quantity, p.notes || '']);
  if (p.action === 'delete')
    sheet.deleteRow(p.id);
  if (p.action === 'update') {
    const row = p.id;
    sheet.getRange(row, 1, 1, 5).setValues([[p.name, p.location, p.category, p.quantity, p.notes || '']]);
  }
  return ContentService.createTextOutput('ok');
}
