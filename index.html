// Sheet layout (columns A–L):
// Cage W: Item(A), Shelf(B), Note(C)
// Cage X: Item(D), Note(E)
// Cage Y: Item(F), Note(G)
// General Cage: Item(H), Note(I)
// UPAC Room 3410: Item(J), SubLocation(K), Note(L)
// Row 1 = headers, data starts row 2

const SECTIONS = [
  { name: 'Cage W',         cols: 3, fields: ['name','shelf','note']        },
  { name: 'Cage X',         cols: 2, fields: ['name','note']                },
  { name: 'Cage Y',         cols: 2, fields: ['name','note']                },
  { name: 'General Cage',   cols: 2, fields: ['name','note']                },
  { name: 'UPAC Room 3410', cols: 3, fields: ['name','sublocation','note']  },
];
const DATA_START_ROW = 2;

// Pre-compute starting column (1-based) for each section
function getSectionStartCols() {
  let col = 1;
  return SECTIONS.map(s => { const c = col; col += s.cols; return c; });
}

function getSheet() {
  return SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
}

function readAllItems() {
  const sheet = getSheet();
  const lastRow = sheet.getLastRow();
  if (lastRow < DATA_START_ROW) return [];
  const startCols = getSectionStartCols();
  const totalCols = startCols[startCols.length - 1] + SECTIONS[SECTIONS.length - 1].cols - 1;
  const numRows = lastRow - DATA_START_ROW + 1;
  const data = sheet.getRange(DATA_START_ROW, 1, numRows, totalCols).getValues();
  const items = [];
  SECTIONS.forEach((sec, si) => {
    const colOffset = startCols[si] - 1; // 0-based
    data.forEach((row, ri) => {
      const name = row[colOffset];
      if (name && name.toString().trim() !== '') {
        const item = { id: si + '_' + ri, section: si, rowIndex: ri, location: sec.name };
        sec.fields.forEach((f, fi) => {
          item[f] = row[colOffset + fi] ? row[colOffset + fi].toString().trim() : '';
        });
        items.push(item);
      }
    });
  });
  return items;
}

function getRowValues(si, p) {
  return SECTIONS[si].fields.map(f => p[f] || '');
}

function findFirstEmptyRow(sheet, col, startRow, lastRow) {
  if (lastRow < startRow) return startRow;
  const colData = sheet.getRange(startRow, col, lastRow - startRow + 1, 1).getValues();
  for (let i = 0; i < colData.length; i++) {
    if (!colData[i][0] || colData[i][0].toString().trim() === '') return startRow + i;
  }
  return startRow + colData.length;
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify(readAllItems())).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = getSheet();
  const p = JSON.parse(e.postData.contents);
  const startCols = getSectionStartCols();

  if (p.action === 'add') {
    const si = SECTIONS.findIndex(s => s.name === p.location);
    if (si === -1) return errOut('Unknown location: ' + p.location);
    const col = startCols[si];
    const targetRow = findFirstEmptyRow(sheet, col, DATA_START_ROW, sheet.getLastRow());
    sheet.getRange(targetRow, col, 1, SECTIONS[si].cols).setValues([getRowValues(si, p)]);
  }

  if (p.action === 'update') {
    const si = parseInt(p.section);
    const col = startCols[si];
    const row = DATA_START_ROW + parseInt(p.rowIndex);
    sheet.getRange(row, col, 1, SECTIONS[si].cols).setValues([getRowValues(si, p)]);
  }

  if (p.action === 'delete') {
    const si = parseInt(p.section);
    const col = startCols[si];
    const row = DATA_START_ROW + parseInt(p.rowIndex);
    sheet.getRange(row, col, 1, SECTIONS[si].cols).clearContent();
  }

  return ContentService.createTextOutput(JSON.stringify(readAllItems())).setMimeType(ContentService.MimeType.JSON);
}

function errOut(msg) {
  return ContentService.createTextOutput(JSON.stringify({ error: msg })).setMimeType(ContentService.MimeType.JSON);
}
