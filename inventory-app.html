Connect to Google Sheets

This lets everyone share the same live data. You only need to do this once.

  1. Open Google Sheets and create a new sheet with columns: Name, Location, Category, Quantity, Notes
  2. Go to Extensions → Apps Script, paste the web app script (see instructions below), and click Deploy → Web app → Anyone.
  3. Copy the web app URL and paste it below.
Show Apps Script code to paste 
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1).map((r,i)=>({id:i+2,...Object.fromEntries(headers.map((h,j)=>[h.toLowerCase(),r[j]]))}));
  return ContentService.createTextOutput(JSON.stringify(rows)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const p = JSON.parse(e.postData.contents);
  if(p.action==='add') sheet.appendRow([p.name,p.location,p.category,p.quantity,p.notes||'']);
  if(p.action==='delete') sheet.deleteRow(p.id);
  return ContentService.createTextOutput('ok');
}
